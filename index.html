async function fetchTelemetry() { try { // Direct fetch using GitHub's cache-busting timestamp const response = await fetch('./alert.json?t=' + Date.now(), { cache: 'no-store' }); if (!response.ok) throw new Error('Failed to load telemetry payload'); const data = await response.json(); // Update Card DOM Elements document.getElementById('status').innerText = data.status || 'UNKNOWN'; document.getElementById('emg-raw').innerText = data.emg_raw !== undefined ? data.emg_raw : '0'; document.getElementById('emg-volts').innerText = data.emg_volts !== undefined ? Number(data.emg_volts).toFixed(2) : '0.00'; document.getElementById('accel').innerText = data.accel_g !== undefined ? Number(data.accel_g).toFixed(2) : '0.00'; document.getElementById('gyro').innerText = data.gyro_dps !== undefined ? data.gyro_dps : '0'; document.getElementById('hr').innerText = data.heart_rate !== undefined ? data.heart_rate : '0'; // Update Status Banner Styling const statusCard = document.getElementById('status-card'); const statusText = document.getElementById('status'); if (data.status === 'ALERT') { statusCard.classList.add('alert-active'); } else { statusCard.classList.remove('alert-active'); statusText.style.color = 'var(--status-nominal)'; } // Add Timestamp and Data to Chart const timeLabel = new Date().toLocaleTimeString(); telemetryChart.data.labels.push(timeLabel); telemetryChart.data.datasets[0].data.push(data.emg_volts || 0); telemetryChart.data.datasets[1].data.push(data.heart_rate || 0); // Limit the number of visible points to scroll the chart smoothly if (telemetryChart.data.labels.length > MAX_DATA_POINTS) { telemetryChart.data.labels.shift(); telemetryChart.data.datasets[0].data.shift(); telemetryChart.data.datasets[1].data.shift(); } telemetryChart.update(); document.getElementById('last-updated').innerText = timeLabel; } catch (error) { console.error('Telemetry fetch error:', error); } }
